cis practical 5 Building a Simple Devanagari Character Classifier using KNN

Cell 1 — Import libraries

import cv2
import os
import numpy as np
import matplotlib.pyplot as plt

from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report



Cell 2 — Read the dataset paths

train_path = r"C:\Users\hp\Downloads\devanagari+handwritten+character+dataset (1)\DevanagariHandwrittenCharacterDataset\Train"

test_path = train_path.replace("Train", "Test")




Cell 3 — Load training and testing images and converting to numpy arrays

def load_data(path, limit=100):
    X, y = [], []

    for c in sorted(os.listdir(path)):
        folder = os.path.join(path, c)

        if os.path.isdir(folder):
            for file in sorted(os.listdir(folder))[:limit]:
                img = cv2.imread(os.path.join(folder, file), 0)

                if img is not None:
                    img = cv2.resize(img, (32, 32))
                    X.append(img.flatten())
                    y.append(c)

    return np.array(X), np.array(y)


X_train, y_train = load_data(train_path)
X_test, y_test = load_data(test_path)

print("Training shape:", X_train.shape)
print("Testing shape:", X_test.shape)


Cell 4 — Normalize pixel values

X_train = X_train / 255.0
X_test = X_test / 255.0



Cell 5 — Train the KNN model

model = KNeighborsClassifier(n_neighbors=5)
model.fit(X_train, y_train)

print("KNN Model Trained")



Cell 6 — Predict characters

prediction = model.predict(X_test)

print("Actual:", y_test[:10])
print("Predicted:", prediction[:10])



Cell 7 — Calculate accuracy

accuracy = accuracy_score(y_test, prediction)

print("Accuracy:", accuracy)



Cell 8 — Generate the confusion matrix

cm = confusion_matrix(y_test, prediction, labels=sorted(os.listdir(train_path)))
print(cm)



Cell 9 — Visualize the confusion matrix

plt.figure(figsize=(8, 6))
plt.imshow(cm)
plt.title("KNN Confusion Matrix")
plt.xlabel("Predicted Character")
plt.ylabel("Actual Character")
plt.colorbar()
plt.show()



Cell 10 — Classification report

print(classification_report(y_test, prediction, zero_division=0))



Cell 11 — Predict a new character

folder = os.path.join(test_path, sorted(os.listdir(train_path))[0])
file = os.listdir(folder)[0]
img_path = os.path.join(folder, file)

img = cv2.imread(img_path, 0)
img = cv2.resize(img, (32, 32))
img = img.flatten().reshape(1, -1) / 255.0

result = model.predict(img)
print("Predicted Character:", result[0])
